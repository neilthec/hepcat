cmake_minimum_required(VERSION 3.16)
project(HEPCATAllPositiveFormulaChecks LANGUAGES CXX)
set(SPINAS_ROOT "" CACHE PATH "SPINAS source, build or installation directory")
find_path(SPINAS_INCLUDE_DIR spinas.h HINTS "${SPINAS_ROOT}/include" REQUIRED)
find_library(SPINAS_LIBRARY NAMES spinas HINTS "${SPINAS_ROOT}/build" "${SPINAS_ROOT}/lib" REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(HEPCAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(qed-all-positive ../all-positive-validation.cpp)
target_compile_features(qed-all-positive PRIVATE cxx_std_17)
get_filename_component(SPINAS_SOURCE_DIR "${SPINAS_INCLUDE_DIR}" DIRECTORY)
target_include_directories(qed-all-positive PRIVATE "${SPINAS_SOURCE_DIR}" "${SPINAS_INCLUDE_DIR}" "${HEPCAT_ROOT}/source/internal/export")
target_link_libraries(qed-all-positive PRIVATE "${SPINAS_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${SPINAS_INCLUDE_DIR}")
set(CMAKE_REQUIRED_LIBRARIES "${SPINAS_LIBRARY}")
set(ABI_PROBE "#include <spinas.h>\nint main(){spinas::particle p(ldouble(1));return p.get_mass()>0?0:1;}")
unset(SPINAS_DOUBLE_ABI CACHE)
check_cxx_source_compiles("${ABI_PROBE}" SPINAS_DOUBLE_ABI)
if(NOT SPINAS_DOUBLE_ABI)
  set(CMAKE_REQUIRED_DEFINITIONS -DWITH_LONG_DOUBLE)
  unset(SPINAS_LONG_DOUBLE_ABI CACHE)
  check_cxx_source_compiles("${ABI_PROBE}" SPINAS_LONG_DOUBLE_ABI)
  if(NOT SPINAS_LONG_DOUBLE_ABI)
    message(FATAL_ERROR "The SPINAS headers and library do not agree on scalar precision.")
  endif()
  target_compile_definitions(qed-all-positive PRIVATE WITH_LONG_DOUBLE)
endif()
